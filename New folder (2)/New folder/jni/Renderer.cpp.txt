#include <GLES2/gl2.h>
#include <GLES2/gl2ext.h>

/**
 * Renderer.cpp

void drawFrame(float* pixels, int width, int height) {

    glClear(GL_COLOR_BUFFER_BIT);
    glDrawArrays(GL_TRIANGLES, 0, 3);
}
